local function safeSet(instance, property, value)
    if not instance then return false end
    return pcall(function() instance[property] = value end)
end
for _, key in ipairs(lightingKeys) do
    local ok, value = pcall(function() return Lighting[key] end)
    if ok then originalLighting[key] = value end
end
local firstCamera = workspace.CurrentCamera
local firstCameraFOV = firstCamera and firstCamera.FieldOfView or 70

local function connect(signal, callback)
    local connection = signal:Connect(callback)
    table.insert(connections, connection)
    return connection
end

local function setTween(instance, duration, props, easing)
    if not running or not instance then return end
    if runningTweens[instance] then
        pcall(function() runningTweens[instance]:Cancel() end)
    end
    local ok, result = pcall(function()
        local item = TweenService:Create(instance, TweenInfo.new(duration, easing or Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props)
        item:Play()
        return item
    end)
    if ok then runningTweens[instance] = result end
    return ok and result or nil
end

local accented = {}
local function trackAccent(object, property)
    table.insert(accented, {object = object, property = property})
    safeSet(object, property, accent())
    return object
end
local function refreshAccent()
    for _, entry in ipairs(accented) do
        if entry.object and entry.object.Parent then
            safeSet(entry.object, entry.property, accent())
        end
    end
end

local function restoreLighting()
    for property, value in pairs(originalLighting) do safeSet(Lighting, property, value) end
end
local function makeEffect(id, className)
    local old = spawnedEffects[id]
    if old and old.Parent then return old end
    local ok, created = pcall(function()
        local item = Instance.new(className)
        item.Name = "AURA_V2_" .. id
        item.Parent = Lighting
        return item
    end)
    if ok then spawnedEffects[id] = created; return created end
    warn("[AURA] Cannot create effect " .. tostring(className))
    return nil
end
local function toggleEffect(id, className, enabled)
    if not enabled then
        local instance = spawnedEffects[id]
        if instance then
            if className == "Atmosphere" then
                pcall(function() instance:Destroy() end)
                spawnedEffects[id] = nil
            else
                safeSet(instance, "Enabled", false)
            end
        end
        return nil
    end
    local instance = makeEffect(id, className)
    if className ~= "Atmosphere" then safeSet(instance, "Enabled", true) end
    return instance
end

local function applyLighting()
    -- Never repeatedly modify lighting every frame: some games manage it themselves.
    restoreLighting()
    if state.preset ~= "Original" then
        for _, preset in ipairs(PRESETS) do
            if preset.key == state.preset then
                safeSet(Lighting, "ClockTime", preset.time)
                safeSet(Lighting, "Brightness", preset.bright)
                safeSet(Lighting, "ExposureCompensation", preset.exp)
                safeSet(Lighting, "Ambient", preset.ambient)
                safeSet(Lighting, "OutdoorAmbient", preset.outdoor)
                safeSet(Lighting, "FogEnd", preset.fog)
                safeSet(Lighting, "FogColor", preset.fogColor)
                break
            end
        end
    end
    if state.customLighting then
        safeSet(Lighting, "ClockTime", state.time)
        safeSet(Lighting, "Brightness", state.brightness)
        safeSet(Lighting, "ExposureCompensation", state.exposure)
        safeSet(Lighting, "FogEnd", state.fogDistance)
        safeSet(Lighting, "ShadowSoftness", state.shadowSoftness)
    end
end
local function refreshEffects()
    local bloom = toggleEffect("Bloom", "BloomEffect", state.bloom)
    if bloom then
        safeSet(bloom, "Intensity", state.bloomIntensity)
        safeSet(bloom, "Size", math.floor(state.bloomSize + .5))
        safeSet(bloom, "Threshold", 1.1)
    end
    local rays = toggleEffect("SunRays", "SunRaysEffect", state.rays)
    if rays then
        safeSet(rays, "Intensity", state.raysIntensity)
        safeSet(rays, "Spread", .78)
    end
    local color = toggleEffect("Grade", "ColorCorrectionEffect", state.grade)
    if color then
        local warmth = state.tint
        local tint
        if warmth >= 0 then
            tint = Color3.new(1, 1 - warmth * .18, 1 - warmth * .33)
        else
            tint = Color3.new(1 + warmth * .30, 1 + warmth * .10, 1)
        end
        safeSet(color, "TintColor", tint)
        safeSet(color, "Contrast", state.contrast)
        safeSet(color, "Saturation", state.saturation)
        safeSet(color, "Brightness", state.gradeBrightness)
    end
    local depth = toggleEffect("DoF", "DepthOfFieldEffect", state.dof)
    if depth then
        safeSet(depth, "NearIntensity", state.nearBlur)
        safeSet(depth, "FarIntensity", state.farBlur)
        safeSet(depth, "FocusDistance", state.focusDistance)
        safeSet(depth, "InFocusRadius", 40)
    end
    local haze = toggleEffect("Atmosphere", "Atmosphere", state.haze)
    if haze then
        safeSet(haze, "Density", state.hazeDensity)
        safeSet(haze, "Offset", 0)
        safeSet(haze, "Haze", 1.2)
        safeSet(haze, "Glare", state.hazeGlare)
        safeSet(haze, "Color", Color3.fromRGB(198, 215, 245))
        safeSet(haze, "Decay", Color3.fromRGB(112, 133, 175))
    end
end
local function applyFOV()
    local camera = workspace.CurrentCamera
    if not camera then return end
    local nextFov = state.customFOV and state.fov or firstCameraFOV
    if state.smoothFOV and not state.reduceMotion then
        setTween(camera, .35, {FieldOfView = nextFov})
    else
        safeSet(camera, "FieldOfView", nextFov)
    end
end

local function make(class, props, parent)
    local obj = Instance.new(class)
    for key, value in pairs(props or {}) do obj[key] = value end
    obj.Parent = parent
    return obj
end
local function corner(parent, radius)
    return make("UICorner", {CornerRadius=UDim.new(0,radius)}, parent)
end
local function border(parent, color, alpha, thickness)
    return make("UIStroke", {Color=color or shades.border, Transparency=alpha or .75, Thickness=thickness or 1}, parent)
end
local function text(parent, content, fontSize, bold, color, props)
    local options = {
        BackgroundTransparency=1, Text=content,
        TextColor3=color or shades.white,
        Font=bold and Enum.Font.GothamBold or Enum.Font.Gotham,
        TextSize=fontSize, TextXAlignment=Enum.TextXAlignment.Left,
        TextYAlignment=Enum.TextYAlignment.Center,
        TextTruncate=Enum.TextTruncate.AtEnd,
    }
    for k,v in pairs(props or {}) do options[k]=v end
    return make("TextLabel", options, parent)
end
local function button(parent, title, size)
    local b = make("TextButton", {
        Size=size, BackgroundColor3=shades.card, BackgroundTransparency=.12,
        Text=title, TextColor3=shades.white, TextSize=12, Font=Enum.Font.GothamBold,
        BorderSizePixel=0, AutoButtonColor=false,
    }, parent)
    corner(b, 11)
    border(b, shades.border, .82)
    connect(b.MouseEnter, function()
        setTween(b, state.reduceMotion and 0 or .18, {BackgroundTransparency=.01})
    end)
    connect(b.MouseLeave, function()
        setTween(b, state.reduceMotion and 0 or .18, {BackgroundTransparency=.12})
    end)
    return b
end

local function switchPage(name)
    if not pages[name] then return end
    if not pageBuilt[name] then
        pageBuilt[name] = true
        local build = pageBuilders[name]
        if build then
            local ok, err = pcall(build)
            if not ok then warn("[AURA] Page " .. name .. ": " .. tostring(err)) end
        end
    end
    state.page=name
    for pageName, scroller in pairs(pages) do
        if pageName==name then
            scroller.Visible=true
            scroller.CanvasPosition=Vector2.new(0,0)
            if not state.reduceMotion then
                scroller.Position=UDim2.fromOffset(247,151)
                setTween(scroller,.25,{Position=UDim2.fromOffset(234,143)},Enum.EasingStyle.Quint)
            else
                scroller.Position=UDim2.fromOffset(234,143)
            end
        else
            scroller.Visible=false
        end
    end
    title.Text=meta[name][1]
    subtitle.Text=meta[name][2]
    for _, cb in pairs(tabRefresh) do cb() end
end

local function addSlider(pageName,key,headline,description,minValue,maxValue,format)
    local card=makeCard(pageName,90)
    text(card,headline,13,true,shades.white,{Position=UDim2.fromOffset(14,7),Size=UDim2.new(1,-125,0,21)})
    local valueText=trackAccent(text(card,"",12,true,accent(),{Position=UDim2.new(1,-111,0,8),Size=UDim2.fromOffset(96,20),TextXAlignment=Enum.TextXAlignment.Right}),"TextColor3")
    text(card,description,10,false,shades.muted,{Position=UDim2.fromOffset(14,29),Size=UDim2.new(1,-26,0,18)})
    local track=make("Frame",{
        Position=UDim2.fromOffset(18,70),Size=UDim2.new(1,-36,0,5),
        BackgroundColor3=Color3.fromRGB(64,76,96),BorderSizePixel=0,
    },card)
    corner(track,3)
    local fill=trackAccent(make("Frame",{
        Size=UDim2.fromScale(.5,1),BackgroundColor3=accent(),BorderSizePixel=0,
    },track),"BackgroundColor3")
    corner(fill,3)
    local dot=make("Frame",{
        AnchorPoint=Vector2.new(.5,.5),Size=UDim2.fromOffset(15,15),
        Position=UDim2.fromScale(.5,.5),BackgroundColor3=shades.white,BorderSizePixel=0,
    },track)
    corner(dot,10)
    local target=make("TextButton",{
        Position=UDim2.new(0,-6,.5,-12),Size=UDim2.new(1,12,0,24),
        BackgroundTransparency=1,Text="",AutoButtonColor=false,
    },track)
    local dragging=false
    local pointer=nil
    local function redraw()
        local pct=math.clamp((state[key]-minValue)/(maxValue-minValue),0,1)
        fill.Size=UDim2.fromScale(pct,1)
        dot.Position=UDim2.fromScale(pct,.5)
        valueText.Text=format(state[key])
    end
    sliderRefresh[key]=redraw
    local function fromX(x)
        local w=track.AbsoluteSize.X
        if w<1 then return end
        local fraction=math.clamp((x-track.AbsolutePosition.X)/w,0,1)
        state[key]=minValue+(maxValue-minValue)*fraction
        redraw()
        onStateChange(key)
    end
    connect(target.InputBegan,function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
            dragging=true
            pointer=input.UserInputType==Enum.UserInputType.Touch and input or nil
            fromX(input.Position.X)
        end
    end)
    connect(UserInputService.InputChanged,function(input)
        if not dragging then return end
        if (pointer and input==pointer) or (not pointer and input.UserInputType==Enum.UserInputType.MouseMovement) then
            fromX(input.Position.X)
        end
    end)
    connect(UserInputService.InputEnded,function(input)
        if not dragging then return end
        if (pointer and input==pointer) or (not pointer and input.UserInputType==Enum.UserInputType.MouseButton1) then
            dragging=false;pointer=nil
        end
    end)
    redraw()
    return card
end
local function pct(v) return tostring(math.floor(v*100+.5)).."%" end
local function signed(v) return string.format("%+.2f",v) end
local function number1(v) return string.format("%.1f",v) end
local function integer(v) return tostring(math.floor(v+.5)) end
local function refreshUI()
    for _,redraw in pairs(switchRefresh) do redraw(false) end
    for _,redraw in pairs(sliderRefresh) do redraw() end
    for _,redraw in pairs(presetRefresh) do redraw() end
    for _,redraw in pairs(themeRefresh) do redraw() end
    refreshAccent()
end

-- The menu animation is independent of the graphical post-processing effects.
local viewportScale=1
local function recalcScale()
    local camera=workspace.CurrentCamera
    if not camera then return end
    local v=camera.ViewportSize
    viewportScale=math.clamp(math.min(1,(v.X-30)/814,(v.Y-32)/543),.43,1)
    if not state.opened then panelScale.Scale=viewportScale*state.uiScale*.96 end
end
local function applyBackdrop()
    panel.BackgroundTransparency=state.uiOpacity
    if state.opened then
        if state.dimBackground then
            overlay.Visible=true
            local target=.69
            if state.reduceMotion then overlay.BackgroundTransparency=target
            else setTween(overlay,.25,{BackgroundTransparency=target}) end
        else
            overlay.BackgroundTransparency=1
            overlay.Visible=false
        end
    else
        overlay.BackgroundTransparency=1
    end
    local amount=(state.opened and state.menuBlur) and 6 or 0
    if amount>0 then
        local instance=makeEffect("MenuBlur","BlurEffect")
        if instance then
            safeSet(instance,"Enabled",true)
            if state.reduceMotion then safeSet(instance,"Size",amount)
            else setTween(instance,.23,{Size=amount}) end
        end
    else
        local instance=spawnedEffects.MenuBlur
        if instance then
            safeSet(instance,"Size",0)
            safeSet(instance,"Enabled",false)
        end
    end
end
local function setOpened(value)
    if not running then return end
    state.opened=not not value
    local targetScale=viewportScale*state.uiScale
    if state.opened then
        panel.Visible=true
        panelScale.Scale=targetScale*(state.reduceMotion and 1 or .94)
        panel.Position=UDim2.new(.5,0,.5,state.reduceMotion and 0 or 12)
        if state.reduceMotion then
            panelScale.Scale=targetScale
            panel.Position=UDim2.new(.5,0,.5,0)
        else
            setTween(panelScale,.32,{Scale=targetScale},Enum.EasingStyle.Back)
            setTween(panel,.32,{Position=UDim2.new(.5,0,.5,0)})
        end
        applyBackdrop()
    else
        applyBackdrop()
        if state.reduceMotion then
            panel.Visible=false
            overlay.Visible=false
        else
            local tw=setTween(panelScale,.18,{Scale=targetScale*.96})
            setTween(panel,.18,{Position=UDim2.new(.5,0,.5,10)})
            if tw then
                local c
                c=tw.Completed:Connect(function()
                    if c then c:Disconnect() end
                    if running and not state.opened then
                        panel.Visible=false;overlay.Visible=false
                    end
                end)
                table.insert(connections,c)
            else
                panel.Visible=false;overlay.Visible=false
            end
        end
    end
end

    toastSerial=toastSerial+1
    local id=toastSerial

        if not running or id~=toastSerial then return end
        if state.reduceMotion then toastFrame.Visible=false

                    if running and id==toastSerial then toastFrame.Visible=false end
                end)
                table.insert(connections,c)
            end
        end
    end)

    state.preset="Original"
    state.customLighting=false
    state.customFOV=false
    state.bloom=false;state.rays=false;state.grade=false;state.dof=false;state.haze=false
    state.menuBlur=false
    applyLighting()

    applyFOV()
    applyBackdrop()
    footStatus.Text="SAFE MODE READY"
    refreshUI()
end
local function updateStatus()
    local any=state.preset~="Original" or state.customLighting or state.customFOV
        or state.bloom or state.rays or state.grade or state.dof or state.haze
    footStatus.Text=any and "CUSTOM VISUALS ON" or "SAFE MODE READY"
end
onStateChange=function(key)
    if key=="fpsHUD" then
        fpsHUD.Visible=state.fpsHUD
    elseif key=="hideFloatingButton" then
        mini.Visible=not state.hideFloatingButton
    elseif key=="reduceMotion" then
        -- Reduced UI motion is applied immediately to future animations.
        if state.reduceMotion then state.menuBlur=false end
        applyBackdrop()
    elseif key=="uiOpacity" or key=="menuBlur" or key=="dimBackground" then
        applyBackdrop()
    elseif key=="uiScale" then
        panelScale.Scale=viewportScale*state.uiScale
    elseif key=="modelScale" then
        refreshModelSize()
    elseif key=="customLighting" or key=="brightness" or key=="exposure" or key=="time" or key=="fogDistance" or key=="shadowSoftness" then
        if key~="customLighting" then state.customLighting=true end
        applyLighting()
    elseif key=="customFOV" or key=="fov" or key=="smoothFOV" then
        if key=="fov" then state.customFOV=true end
        applyFOV()
    elseif key=="bloom" or key=="bloomIntensity" or key=="bloomSize" or key=="rays" or key=="raysIntensity"
        or key=="grade" or key=="saturation" or key=="contrast" or key=="tint" or key=="gradeBrightness"
        or key=="dof" or key=="nearBlur" or key=="farBlur" or key=="focusDistance"
        or key=="haze" or key=="hazeDensity" or key=="hazeGlare" then
        if key=="bloomIntensity" or key=="bloomSize" then state.bloom=true end
        if key=="raysIntensity" then state.rays=true end
        if key=="saturation" or key=="contrast" or key=="tint" or key=="gradeBrightness" then state.grade=true end
        if key=="nearBlur" or key=="farBlur" or key=="focusDistance" then state.dof=true end
        if key=="hazeDensity" or key=="hazeGlare" then state.haze=true end
        refreshEffects()
    end
    updateStatus()
    refreshUI()
end

local function recolorForPart(name,theme)

local function freshVisualPart(parent, name, color, material, size, transparency)
    return make("Part",{
        Name=name,Size=size,Color=color,Material=material,
        Anchored=true,CanCollide=false,CanTouch=false,CanQuery=false,
        CastShadow=false,Massless=true,Transparency=transparency or 0,
        TopSurface=Enum.SurfaceType.Smooth,BottomSurface=Enum.SurfaceType.Smooth,
    },parent)
end
-- Each cosmetic part is welded to a limb (no per-frame world-space teleport).
-- No server state, collisions or movement properties are changed.
local function attachVisual(src, part, offset)
    if not src or not part or not src:IsA("BasePart") then return nil end
    local transform=offset or CFrame.new()
    part.CFrame=src.CFrame*transform
    local joint=Instance.new("Weld")
    joint.Name="AuraCosmeticWeld"
    joint.Part0=src
    joint.Part1=part
    joint.C0=transform
    joint.C1=CFrame.new()
    joint.Parent=part
    part.Anchored=false
    return joint
end

local function eraseVisual(uid)
    local entry=visuals[uid]
    if not entry then return end
    visuals[uid]=nil
    for source,old in pairs(entry.hidden) do
        if source and source.Parent then
            pcall(function() source.LocalTransparencyModifier=old end)
        end
    end
    if entry.container then pcall(function()entry.container:Destroy()end) end
end

local function buildVisual(target, skinKey, wingsKey, modelKey)
    if not running then return end
    local uid=target.UserId
    eraseVisual(uid)
    modelKey=(MODEL_SPECS[modelKey] and modelKey) or "None"
    if skinKey=="Original" and wingsKey=="None" and modelKey=="None" then return end
    local character=target.Character
    if not character or not character.Parent then return end
    local theme=SKINS[skinKey] or SKINS.Original
    local wingTheme=WINGS[wingsKey] or WINGS.None
    local bodySource=character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso")
    if not bodySource then return end
    local container=make("Model",{Name="AURA_LOOK_"..tostring(uid)},cosmeticFolder)
    local entry={container=container,hidden={},parts={},details={},wingParts={},char=character,target=target,wing=wingsKey,skin=skinKey,model=modelKey}
    visuals[uid]=entry

    if skinKey~="Original" or modelKey~="None" then
        -- Hide the original avatar only on this particular client.
        for _,obj in ipairs(character:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name~="HumanoidRootPart" then
                entry.hidden[obj]=obj.LocalTransparencyModifier
                pcall(function() obj.LocalTransparencyModifier=1 end)
            end
        end
        if modelKey~="None" then
            addWearableArmor(entry,character,modelKey,false)
        else
        -- Procedural body replacement. Works with most R6/R15 avatars.
        for _,source in ipairs(character:GetDescendants()) do
            if source:IsA("BasePart") and source.Name~="HumanoidRootPart" and source.Parent==character then
                local isHead=source.Name=="Head"
                local limb=freshVisualPart(container,"Skin_"..source.Name,recolorForPart(source.Name,theme),theme.material,
                    Vector3.new(source.Size.X*1.02,source.Size.Y*1.02,source.Size.Z*1.02),1)
                if isHead and skinKey=="Astral" then limb.Shape=Enum.PartType.Ball end
                entry.parts[#entry.parts+1]={src=source,part=limb}
                attachVisual(source,limb,CFrame.new())
                if isHead then
                    local visor=freshVisualPart(container,"Visor",theme.secondary,Enum.Material.Neon,
                        Vector3.new(math.max(.5,source.Size.X*.74),math.max(.1,source.Size.Y*.13),.105),1)
                    local off=CFrame.new(0,.02,-source.Size.Z*.53)
                    entry.details[#entry.details+1]={src=source,part=visor,offset=off}
                    attachVisual(source,visor,off)
                elseif source.Name=="Torso" or source.Name=="UpperTorso" then
                    local plate=freshVisualPart(container,"CorePlate",theme.secondary,Enum.Material.Neon,
                        Vector3.new(source.Size.X*.55,source.Size.Y*.13,.1),1)
                    local off=CFrame.new(0,.17,-source.Size.Z*.53)
                    entry.details[#entry.details+1]={src=source,part=plate,offset=off}
                    attachVisual(source,plate,off)
                end
            end
        end
        end -- model versus skin body shell
    end
    if wingsKey~="None" then
        -- Animated floating feathers/panels: no physical collision, no server state.
        for side=-1,1,2 do
            for j=1,6 do
                local material=wingTheme.glow and Enum.Material.Neon or Enum.Material.SmoothPlastic
                if wingsKey=="Crystal" then material=Enum.Material.Glass end
                local feather=freshVisualPart(container,"Wing_"..side.."_"..j,wingTheme.color,material,
                    Vector3.new(.22,.82+j*.09,.27+j*.07),1)
                if wingsKey=="Butterfly" then
                    feather.Size=Vector3.new(.13,.5+j*.16,.65)
                end
                local spread=state.wingSpread
                local x=side*(.73+j*.23*spread)
                local y=.52+math.sin(j*.55)*.27-j*.09
                local z=.57+j*.05
                local offset=CFrame.new(x,y,z)*CFrame.Angles(0,side*.36,side*(.35+j*.11))
                local joint=attachVisual(bodySource,feather,offset)
                entry.wingParts[#entry.wingParts+1]={part=feather,torso=bodySource,side=side,index=j,joint=joint}
            end
        end
    end
    -- Start at the exact limb positions before any fading begins.
    for _,pair in ipairs(entry.parts) do
        if pair.src and pair.src.Parent then pair.part.CFrame=pair.src.CFrame end
    end
    for _,pair in ipairs(entry.details) do
        if pair.src and pair.src.Parent then pair.part.CFrame=pair.src.CFrame*pair.offset end
    end
    -- Softly fade the new parts in (short tweens to avoid start-up load).
    for _,partData in ipairs(entry.parts) do
        setTween(partData.part,state.reduceMotion and 0 or .3,{Transparency=0})
    end
    for _,detail in ipairs(entry.details) do
        setTween(detail.part,state.reduceMotion and 0 or .3,{Transparency=detail.alpha or 0})
    end
    for _,wingData in ipairs(entry.wingParts) do
        setTween(wingData.part,state.reduceMotion and 0 or .4,{Transparency=0})
    end
end
local function updateLocalLook()
    ownLooks[player.UserId]={skin=state.skin,wings=state.wings,model=state.model}
    buildVisual(player,state.skin,state.wings,state.model)
    redrawLookButtons()
    if shareCodeField then shareCodeField.Text="AURA LOOK "..state.skin.." "..state.wings.." "..state.model end
end
local function assignRemoteLook(target,skinKey,wingsKey,modelKey)
    if target==player then return end
    if not SKINS[skinKey] or not WINGS[wingsKey] or not isValidModelKey(modelKey or "None") then return end
    ownLooks[target.UserId]={skin=skinKey,wings=wingsKey,model=modelKey or "None"}
    -- Limit remote cosmetics for lower-end laptops.
    local already=false
    for _,uid in ipairs(remoteOrder) do if uid==target.UserId then already=true break end end
    if not already then table.insert(remoteOrder,target.UserId) end
    while #remoteOrder>5 do
        local old=table.remove(remoteOrder,1)
        if old~=player.UserId then ownLooks[old]=nil;eraseVisual(old) end
    end
    buildVisual(target,skinKey,wingsKey,modelKey or "None")
end

-- Opt-in visible-chat sharing. It does NOT bypass chat restrictions or silently transmit data.
-- To share a look, copy the code and send it through normal chat manually.
local function onSharedChat(message)
    if not running or not state.acceptSharedLooks then return end
    local textValue=message and message.Text
    local origin=message and message.TextSource
    if type(textValue)~="string" or not origin then return end
    if origin.UserId==player.UserId then return end
    local skin,wing,modelKey=textValue:match("^AURA LOOK ([%a]+) ([%a]+) ([%a]+)$")
    if not skin then
        skin,wing=textValue:match("^AURA LOOK ([%a]+) ([%a]+)$")
        modelKey="None"
    end
    if not skin or not wing or not SKINS[skin] or not WINGS[wing] then return end
    local target=Players:GetPlayerByUserId(origin.UserId)
    if target then assignRemoteLook(target,skin,wing,modelKey) end
end
pcall(function()connect(TextChatService.MessageReceived,onSharedChat)end)
connect(Players.PlayerRemoving,function(target)
    eraseVisual(target.UserId)
    ownLooks[target.UserId]=nil
end)

end

-- No per-frame part.CFrame assignments: welded parts follow R6/R15 automatically.
-- Only a handful of joints are rotated, at 20 Hz, for gentle cosmetic motion.
local cosmeticAccumulator=0
connect(RunService.Heartbeat,function(dt)
    if not running or not next(visuals) then return end
    cosmeticTime=cosmeticTime+dt
    cosmeticAccumulator=cosmeticAccumulator+dt
    if cosmeticAccumulator<1/20 then return end
    cosmeticAccumulator=0
    for uid,entry in pairs(visuals) do
        if not entry.char or not entry.char.Parent then
            eraseVisual(uid)
        else
            local flap=math.sin(cosmeticTime*2.4*state.wingSpeed)*.13
            for _,wing in ipairs(entry.wingParts) do
                if wing.joint and wing.joint.Parent then
                    local side,j=wing.side,wing.index
                    local spread=state.wingSpread
                    local x=side*(.73+j*.23*spread)
                    local y=.52+math.sin(j*.55)*.27-j*.09
                    local z=.57+j*.05
                    wing.joint.C0=CFrame.new(x,y,z)
                        *CFrame.Angles(0,side*(.36+flap),side*(.35+j*.11+flap))
                end
            end
            if state.modelAnimate then
                for _,detail in ipairs(entry.details) do
                    if detail.motion and detail.joint and detail.joint.Parent then
                        detail.joint.C0=detail.baseOffset
                            *CFrame.Angles(0,math.sin(cosmeticTime*1.8)*.08,math.sin(cosmeticTime*1.5)*.07)
                    end
                end
            end
        end
    end
end)

-- V6: the live 3D model is generated by addWearableArmor inside buildVisual.
-- No separate flying model instances exist. All suit parts follow character limbs.
refreshModelSize=function()
    local entry=visuals[player.UserId]
    if not entry or entry.model=="None" then return end
    for _,item in ipairs(entry.parts) do
        if item.wearable and item.part and item.part.Parent then
            item.part.Size=item.baseSize*state.modelScale
        end
    end
    for _,item in ipairs(entry.details) do
        if item.wearable and item.part and item.part.Parent then
            item.part.Size=item.baseSize*state.modelScale
        end
    end
end

local frames,windowStart=0,os.clock()
connect(RunService.RenderStepped,function()
    if not running then return end
    frames=frames+1
    local now=os.clock()
    local dt=now-windowStart
    if dt<.7 then return end
    local fps=math.floor(frames/dt+.5)
    frames=0;windowStart=now
    local value="FPS  "..fps
    fpsFooter.Text=value
    fpsHUDValue.Text=value
    dockFPS.Text="FPS "..fps
    local pingText="PING --"
    local okPing, pingResult=pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValueString()
    end)
    if okPing and pingResult then
        pingText="PING "..tostring(pingResult):gsub("%s+", "")
    else
        local okNative, nativePing=pcall(function()return player:GetNetworkPing()end)
        if okNative and type(nativePing)=="number" then
            pingText="PING "..tostring(math.floor(nativePing*1000+.5)).."ms"
        end
    end
    dockPing.Text=pingText
    if diagFPS then diagFPS.Text="FPS: "..fps end
    local cam=workspace.CurrentCamera
    if cam then
        local sz=cam.ViewportSize
        if diagViewport then diagViewport.Text="Display: "..math.floor(sz.X).." × "..math.floor(sz.Y) end
    end
    local active=0
    for _,which in ipairs({"bloom","rays","grade","dof","haze"}) do
        if state[which] then active=active+1 end
    end
    if diagEffects then diagEffects.Text="Active effects: "..active end
    if diagStatus then diagStatus.Text="Status: "..(state.preset=="Original" and "Original" or state.preset) end
end)
