local function own(signal, fn)
    local conn = signal:Connect(fn)
    table.insert(connections, conn)
    return conn
end

local function tween(object, time, props, style)
    if not alive or not object then return nil end
    if activeTweens[object] then pcall(function() activeTweens[object]:Cancel() end) end
    local item = TweenService:Create(object, TweenInfo.new(time, style or Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props)
    activeTweens[object] = item
    item:Play()
    return item
end

local function create(className, properties, parent)
    local obj = Instance.new(className)
    for key, value in pairs(properties or {}) do obj[key] = value end
    obj.Parent = parent
    return obj
end
local function rounded(obj, n)
    return create("UICorner", {CornerRadius = UDim.new(0, n)}, obj)
end
local function stroke(obj, color, transparency, thickness)
    return create("UIStroke", {
        Color = color or C.border,
        Transparency = transparency or .65,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, obj)
end
local function gradient(obj, a, b, rotation)
    return create("UIGradient", {Color = ColorSequence.new(a, b), Rotation = rotation or 0}, obj)
end
local function label(parent, content, size, bold, color, props)
    local defaults = {
        BackgroundTransparency = 1,
        Text = content,
        Font = bold and Enum.Font.GothamBold or Enum.Font.Gotham,
        TextSize = size,
        TextColor3 = color or C.white,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }
    for k, v in pairs(props or {}) do defaults[k] = v end
    return create("TextLabel", defaults, parent)
end

local values = {"ClockTime", "Brightness", "Ambient", "OutdoorAmbient", "ExposureCompensation", "FogEnd", "FogStart", "FogColor", "GlobalShadows"}
for _, property in ipairs(values) do
    local ok, value = pcall(function() return Lighting[property] end)
    if ok then originals[property] = value end
end

local function restoreLighting()
    for property, value in pairs(originals) do
        pcall(function() Lighting[property] = value end)
    end
end

local function effect(class, name)
    local instance = Instance.new(class)
    instance.Name = "AURA_VISION_" .. name
    instance.Parent = Lighting
    table.insert(customEffects, instance)
    return instance
end

local function applyPreset(name)
    if name == "Original" then
        restoreLighting()
        color.TintColor = Color3.new(1, 1, 1)
        state.preset = name
        return
    end
    local p = presets[name]
    if not p then return end
    -- Changes here only affect the client's view in supported contexts.
    Lighting.ClockTime = p.clock
    Lighting.Brightness = p.bright
    Lighting.Ambient = p.ambient
    Lighting.OutdoorAmbient = p.outdoor
    Lighting.ExposureCompensation = p.exposure
    Lighting.FogEnd = p.fog
    Lighting.FogColor = p.fogColor
    Lighting.GlobalShadows = true
    color.TintColor = p.tint
    state.preset = name
end

        state[key] = not state[key]
        updateEffects()

local function sliderRow(name, description, key, minVal, maxVal, order, formatter)
    local host = pages.Effects
    local row = create("Frame", {LayoutOrder=order, Size=UDim2.new(1,-3,0,76), BackgroundColor3=C.card, BackgroundTransparency=.26, BorderSizePixel=0}, host)
    rounded(row, 13)
    stroke(row, C.border, .91, 1)
    label(row, name, 13, true, C.white, {Position=UDim2.fromOffset(14, 7), Size=UDim2.new(1,-105,23)})
    local readout = label(row, "", 12, true, C.cyan, {Position=UDim2.new(1,-92,0,9), Size=UDim2.fromOffset(77,20), TextXAlignment=Enum.TextXAlignment.Right})
    label(row, description, 10, false, C.muted, {Position=UDim2.fromOffset(14,29), Size=UDim2.new(1,-25,17)})
    local track = create("Frame", {Position=UDim2.new(0,14,1,-15), Size=UDim2.new(1,-28,0,5), BackgroundColor3=Color3.fromRGB(53,63,85), BorderSizePixel=0, Active=true}, row)
    rounded(track, 3)
    local fill = create("Frame", {Size=UDim2.fromScale(.5,1), BackgroundColor3=C.cyan, BorderSizePixel=0}, track)
    rounded(fill, 3)
    gradient(fill, C.cyan, C.lilac, 0)
    local circle = create("Frame", {AnchorPoint=Vector2.new(.5,.5), Position=UDim2.fromScale(.5,.5), Size=UDim2.fromOffset(14,14), BackgroundColor3=C.white, BorderSizePixel=0}, track)
    rounded(circle, 7)
    local dragging = false
    local touchInput = nil
    local function refresh()
        local pct = math.clamp((state[key]-minVal)/(maxVal-minVal), 0, 1)
        fill.Size = UDim2.fromScale(pct, 1)
        circle.Position = UDim2.fromScale(pct, .5)
        readout.Text = formatter(state[key])
    end
    local function setAt(x)
        if track.AbsoluteSize.X < 1 then return end
        local pct = math.clamp((x - track.AbsolutePosition.X)/track.AbsoluteSize.X, 0, 1)
        state[key] = minVal + (maxVal-minVal)*pct
        updateEffects()
        refresh()
    end
    own(track.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            touchInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
            setAt(input.Position.X)
        end
    end)
    own(UserInputService.InputChanged, function(input)
        if not dragging then return end
        if touchInput and input == touchInput then
            setAt(input.Position.X)
        elseif not touchInput and input.UserInputType == Enum.UserInputType.MouseMovement then
            setAt(input.Position.X)
        end
    end)
    own(UserInputService.InputEnded, function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or (touchInput and input == touchInput)) then
            dragging = false
            touchInput = nil
        end

local desiredScale = 1
local function setOpen(open)
    if not alive then return end
    state.open = open
    if open then
        overlay.Visible = true
        panel.Visible = true
        panel.GroupTransparency = 1
        panel.Position = UDim2.new(.5, 0, .5, 17)
        panelScale.Scale = desiredScale * .92
        tween(panel, .38, {GroupTransparency=0, Position=UDim2.new(.5,0,.5,0)})
        tween(panelScale, .42, {Scale=desiredScale}, Enum.EasingStyle.Back)
        tween(overlay, .25, {BackgroundTransparency=.46})
        tween(blur, .42, {Size=7})
    else
        tween(overlay, .24, {BackgroundTransparency=1})
        tween(blur, .28, {Size=0})
        local fade = tween(panel, .25, {GroupTransparency=1, Position=UDim2.new(.5,0,.5,12)})
        tween(panelScale, .25, {Scale=desiredScale * .94})
        if fade then own(fade.Completed, function()
            if alive and not state.open then
                panel.Visible = false
                overlay.Visible = false
            end
        end) end
    end
end

-- Drag panel by its top bar (mouse or touch).
local draggingPanel = false
local dragStart = nil
local startPosition = nil
local dragTouch = nil
local dragHandle = create("Frame", {Position=UDim2.fromOffset(196,0), Size=UDim2.new(1,-260,0,25), BackgroundTransparency=1, Active=true}, base)
own(dragHandle.InputBegan, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingPanel = true
        dragStart = input.Position
        startPosition = panel.Position
        dragTouch = input.UserInputType == Enum.UserInputType.Touch and input or nil
    end
end)
own(UserInputService.InputChanged, function(input)
    if not draggingPanel then return end
    if (dragTouch and input == dragTouch) or (not dragTouch and input.UserInputType == Enum.UserInputType.MouseMovement) then
        local delta = input.Position - dragStart
        panel.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
    end
end)
own(UserInputService.InputEnded, function(input)
    if (dragTouch and input == dragTouch) or (not dragTouch and input.UserInputType == Enum.UserInputType.MouseButton1) then
        draggingPanel = false
        dragTouch = nil
    end
end)

-- Automatically shrink the panel to fit smaller screens.
local function fitScreen()
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local scale = math.min(1, (viewport.X - 24)/710, (viewport.Y - 28)/482)
    if scale > 0 then
        desiredScale = scale
        panelScale.Scale = state.open and desiredScale or desiredScale * .94
    end
end
fitScreen()
if workspace.CurrentCamera then
    own(workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"), fitScreen)
end

local frameCount, frameWindowStart = 0, os.clock()
own(RunService.RenderStepped, function()
    frameCount = frameCount + 1
    local now = os.clock()
    if now - frameWindowStart >= .5 then
        local fps = math.floor(frameCount / (now-frameWindowStart) + .5)
        fpsText.Text = "FPS  " .. tostring(fps)
        frameCount, frameWindowStart = 0, now
    end
end)

local function cleanup()
    if not alive then return end
    alive = false
    for _, conn in ipairs(connections) do pcall(function() conn:Disconnect() end) end
    for _, item in pairs(activeTweens) do pcall(function() item:Cancel() end) end
    restoreLighting()
    for _, item in ipairs(customEffects) do pcall(function() item:Destroy() end) end
    pcall(function() gui:Destroy() end)
    environment.AURA_VISION_CLEANUP = nil
    environment.AURA_VISION_TOGGLE = nil
end
environment.AURA_VISION_CLEANUP = cleanup
